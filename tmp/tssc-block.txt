    });

    // On phones the empty Code block still owns its own grid rows in the hero, leaving a gap
    // above the title. Hide it and pull the blocks below up by the same number of rows.
    function collapseHost() {
      var host = document.querySelector('.cs-nav-host');
      var grid = host && host.parentElement;
      if (!grid || !grid.classList.contains('fluid-engine')) return;
      grid.querySelectorAll('[data-cs-shift]').forEach(function (x) { x.style.gridRowStart = ''; x.style.gridRowEnd = ''; x.removeAttribute('data-cs-shift'); });
      host.style.display = ''; grid.style.gridTemplateRows = '';
      if (!window.matchMedia('(max-width: 767px)').matches) return;
      var hs = getComputedStyle(host), a = parseInt(hs.gridRowStart, 10), b = parseInt(hs.gridRowEnd, 10);
      if (!(a > 0 && b > a)) return;
      var rows = Array.prototype.filter.call(grid.children, function (x) { return x !== host; }).map(function (x) {
        var c = getComputedStyle(x); return [x, parseInt(c.gridRowStart, 10), parseInt(c.gridRowEnd, 10)];
      });
      // Only collapse when nothing else shares those rows.
      if (rows.some(function (r) { return !(r[1] > 0 && r[2] > 0) || (r[1] < b && r[2] > a); })) return;
      var tracks = getComputedStyle(grid).gridTemplateRows.split(' ').map(parseFloat).filter(function (v) { return !isNaN(v); });
      if (!tracks.length) return;
      var n = b - a, min = Math.min.apply(null, tracks);
      rows.forEach(function (r) {
        if (r[1] >= b) { r[0].style.gridRowStart = r[1] - n; r[0].style.gridRowEnd = r[2] - n; r[0].setAttribute('data-cs-shift', ''); }
      });
      host.style.display = 'none';
      grid.style.gridTemplateRows = 'repeat(' + (tracks.length - n) + ', minmax(' + min + 'px, auto))';
    }
    collapseHost();
    var mq = window.matchMedia('(max-width: 767px)');
    if (mq.addEventListener) mq.addEventListener('change', collapseHost); else if (mq.addListener) mq.addListener(collapseHost);

    // Case study sections: skip the hero, testimonials and contact form.
    var sections = Array.prototype.slice.call(document.querySelectorAll('#sections > section, article section[data-section-id], main section[data-section-id]'))
      .filter(function (s, i, all) { return all.indexOf(s) === i; });
    // Muthabr builds its whole case study inside one Code block: use its chapters instead.
    var chapters = Array.prototype.slice.call(document.querySelectorAll('section.mu-case-chapter'));
    var skipFirst = true;
    if (chapters.length >= 3) { sections = chapters; skipFirst = false; }
    var items = [];
    sections.forEach(function (s, i) {
      if (skipFirst && i === 0) return;
      var h = s.querySelector('.mu-case-kicker') || s.querySelector('h1, h2, h3');
      if (!h) return;
      var text = (s.getAttribute('data-nav-label') || h.textContent).trim().replace(/\.$/, '');
      if (!text || /^[“"]/.test(text) || /^contact us/i.test(text)) return;
      if (!s.id) s.id = 'cs-' + text.toLowerCase().replace(/&/g, 'and').replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
      items.push({ section: s, label: text });
    });
    if (items.length < 3) return;

    var header = document.querySelector('#header, header');
    function headerOffset() { return header ? header.getBoundingClientRect().height : 0; }

    var nav = document.createElement('nav');
    nav.className = 'cs-nav';
    nav.setAttribute('aria-label', 'Case study sections');
    var toggle = document.createElement('button');
    toggle.type = 'button';
    toggle.className = 'cs-nav-toggle';
    toggle.setAttribute('aria-expanded', 'false');
    toggle.textContent = 'Sections';
    var list = document.createElement('ol');
    var seekTimer;
    var links = items.map(function (it) {
      var li = document.createElement('li');
      var a = document.createElement('a');
      a.href = '#' + it.section.id;
      a.innerHTML = '<span class="cs-tick" aria-hidden="true"></span><span class="cs-label"></span>';
      a.querySelector('.cs-label').textContent = it.label;
      a.addEventListener('click', function (e) {
        e.preventDefault();
        var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
        function aim() { window.scrollTo({ top: it.section.getBoundingClientRect().top + window.pageYOffset - headerOffset(), behavior: reduce ? 'auto' : 'smooth' }); }
        aim();
        // Lazy images load during long jumps and push the target down: re-aim once the scroll settles.
        clearInterval(seekTimer);
        var last = -1, tries = 0;
        seekTimer = setInterval(function () {
          if (Math.abs(window.pageYOffset - last) > 1) { last = window.pageYOffset; return; }
          if (Math.abs(it.section.getBoundingClientRect().top - headerOffset()) > 4 && tries++ < 4) { last = -1; aim(); }
          else clearInterval(seekTimer);
        }, 150);
        history.replaceState(null, '', '#' + it.section.id);
        setOpen(false);
      });
      li.appendChild(a);
      list.appendChild(li);
      return a;
    });
    nav.appendChild(list);
    nav.appendChild(toggle);
    document.body.appendChild(nav);

    function setOpen(open) {
      nav.classList.toggle('is-open', open);
      toggle.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
    toggle.addEventListener('click', function () { setOpen(!nav.classList.contains('is-open')); });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape') setOpen(false); });
    document.addEventListener('click', function (e) { if (!nav.contains(e.target)) setOpen(false); });

    // Highlight the section in view; show the nav only between the hero and the end of the case study.
    function update() {
      var mark = window.innerHeight * 0.4;
      var active = -1;
      items.forEach(function (it, i) { if (it.section.getBoundingClientRect().top <= mark) active = i; });
      var last = items[items.length - 1].section.getBoundingClientRect();
      var visible = items[0].section.getBoundingClientRect().top < window.innerHeight * 0.6 && last.bottom > window.innerHeight * 0.4;
      nav.classList.toggle('is-visible', visible);
      if (!visible) setOpen(false);
      links.forEach(function (a, i) {
        a.classList.toggle('is-active', i === active);
        if (i === active) a.setAttribute('aria-current', 'location'); else a.removeAttribute('aria-current');
      });
    }
    var ticking = false;
    window.addEventListener('scroll', function () {
      if (ticking) return;
      ticking = true;
      requestAnimationFrame(function () { ticking = false; update(); });
    }, { passive: true });
    window.addEventListener('resize', update);
    update();
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init);
  else init();
})();
</script>
